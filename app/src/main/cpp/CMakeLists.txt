cmake_minimum_required(VERSION 3.22)
project(piccodec LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(THIRD ${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(WITH_TURBOJPEG OFF CACHE BOOL "" FORCE)
set(WITH_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_JPEG8 ON CACHE BOOL "" FORCE)
set(WITH_SIMD OFF CACHE BOOL "" FORCE)
include(ExternalProject)
ExternalProject_Add(jpeg-build
 SOURCE_DIR ${THIRD}/libjpeg-turbo
 BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/jpeg
 CMAKE_ARGS -DCMAKE_MAKE_PROGRAM=${CMAKE_MAKE_PROGRAM} -DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE} -DANDROID_ABI=${ANDROID_ABI} -DANDROID_PLATFORM=android-26 -DANDROID_STL=c++_static -DCMAKE_POSITION_INDEPENDENT_CODE=ON -DCMAKE_BUILD_TYPE=Release -DENABLE_SHARED=OFF -DENABLE_STATIC=ON -DWITH_TURBOJPEG=OFF -DWITH_TOOLS=OFF -DWITH_TESTS=OFF -DWITH_SIMD=OFF -DWITH_JPEG8=ON
 BUILD_BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/jpeg/libjpeg.a
 INSTALL_COMMAND "")
add_library(jpeg-static STATIC IMPORTED)
set_target_properties(jpeg-static PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_BINARY_DIR}/jpeg/libjpeg.a)
add_dependencies(jpeg-static jpeg-build)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(tiff-tools OFF CACHE BOOL "" FORCE)
set(tiff-tests OFF CACHE BOOL "" FORCE)
set(tiff-contrib OFF CACHE BOOL "" FORCE)
set(tiff-docs OFF CACHE BOOL "" FORCE)
set(tiff-install OFF CACHE BOOL "" FORCE)
set(jpeg OFF CACHE BOOL "" FORCE)
set(zstd OFF CACHE BOOL "" FORCE)
set(webp OFF CACHE BOOL "" FORCE)
set(lzma OFF CACHE BOOL "" FORCE)
set(jbig OFF CACHE BOOL "" FORCE)
set(lerc OFF CACHE BOOL "" FORCE)
add_subdirectory(${THIRD}/libtiff ${CMAKE_CURRENT_BINARY_DIR}/tiff EXCLUDE_FROM_ALL)
file(GLOB_RECURSE RAW_SOURCES ${THIRD}/LibRaw/src/*.cpp)
file(GLOB_RECURSE RAW_PLACEHOLDERS ${THIRD}/LibRaw/src/*_ph.cpp)
list(REMOVE_ITEM RAW_SOURCES ${RAW_PLACEHOLDERS})
add_library(raw STATIC ${RAW_SOURCES})
target_include_directories(raw PUBLIC ${THIRD}/LibRaw PRIVATE ${THIRD}/libjpeg-turbo/src ${CMAKE_CURRENT_BINARY_DIR}/jpeg)
target_compile_definitions(raw PRIVATE LIBRAW_NOTHREADS LIBRAW_NODLL USE_ZLIB USE_JPEG USE_JPEG8)
target_compile_options(raw PRIVATE -O2 -fvisibility=hidden)
target_link_libraries(raw PUBLIC jpeg-static z)
if(ANDROID)
add_library(piccodec SHARED piccodec.cpp)
else()
add_executable(piccodec piccodec.cpp)
target_compile_definitions(piccodec PRIVATE PICCODEC_HOST)
endif()
target_include_directories(piccodec PRIVATE ${THIRD}/libjpeg-turbo/src ${CMAKE_CURRENT_BINARY_DIR}/jpeg ${THIRD}/libtiff/libtiff ${CMAKE_CURRENT_BINARY_DIR}/tiff/libtiff)
target_link_libraries(piccodec raw tiff jpeg-static z)
if(ANDROID)
target_link_libraries(piccodec jnigraphics log)
target_link_options(piccodec PRIVATE -Wl,-z,max-page-size=16384)
endif()
